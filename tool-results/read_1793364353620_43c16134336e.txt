   107→   107→def add_heading(text, style, level=0):
   108→   108→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
   109→   109→    p = Paragraph('<a name="%s"/><b>%s</b>' % (key, text), style)
   110→   110→    p.bookmark_name = key
   111→   111→    p.bookmark_level = level
   112→   112→    p.bookmark_text = text
   113→   113→    p.bookmark_key = key
   114→   114→    return p

   116→   116→def h1_block(text):
   117→   117→    """H1 com regua de acento e protecao contra titulo orfao."""
   118→   118→    rule = HRFlowable(width='100%', thickness=1.2, color=ACCENT,
   119→   119→                      spaceBefore=2, spaceAfter=14)

   125→   125→def safe_keep(elements):
   126→   126→    total = 0
   127→   127→    for el in elements:
   128→   128→        w, hh = el.wrap(AVAIL_W, PAGE_H)
   129→   129→        total += hh
   130→   130→    if total <= MAX_KEEP:
   131→   131→        return [KeepTogether(elements)]
   132→   132→    elif len(elements) >= 2:
   133→   133→        return [KeepTogether(elements[:2])] + list(elements[2:])
   134→   134→    return list(elements)

   136→   136→def embed_image(path, max_width=None, max_height=None):
   137→   137→    if max_width is None:
   138→   138→        max_width = AVAIL_W
   139→   139→    if max_height is None:
   140→   140→        max_height = PAGE_H * 0.35
   141→   141→    pil = PILImage.open(path)
   142→   142→    ow, oh = pil.size
   143→   143→    rw = max_width / ow if ow > max_width else 1.0
   144→   144→    rh = max_height / oh if oh > max_height else 1.0
   145→   145→    r = min(rw, rh)
   146→   146→    return Image(path, width=ow * r, height=oh * r)

   153→   153→def callout_strip(items):
   154→   154→    """items: lista de (numero_grande, rotulo). Gera faixa com 3 cartoes."""
   155→   155→    n = len(items)
   156→   156→    gap = 10
   157→   157→    w = (AVAIL_W - gap * (n - 1)) / n
   158→   158→    row_big, row_lbl = [], []
   159→   159→    col_widths = []
   160→   160→    for i, (big, lbl) in enumerate(items):
   161→   161→        row_big.append(Paragraph('<b>%s</b>' % big, stat_big))
   162→   162→        row_lbl.append(Paragraph(lbl, stat_lbl))
   163→   163→        col_widths.append(w)
   164→   164→        if i < n - 1:
   165→   165→            row_big.append('')
   166→   166→            row_lbl.append('')
   167→   167→            col_widths.append(gap)
   168→   168→    t = Table([row_big, row_lbl], colWidths=col_widths, hAlign='CENTER')
   169→   169→    style = [('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   170→   170→             ('TOPPADDING', (0, 0), (-1, 0), 10),
   171→   171→             ('BOTTOMPADDING', (0, 1), (-1, 1), 10),
   172→   172→             ('TOPPADDING', (0, 1), (-1, 1), 2),
   173→   173→             ('BOTTOMPADDING', (0, 0), (-1, 0), 2),
   174→   174→             ('LEFTPADDING', (0, 0), (-1, -1), 6),
   175→   175→             ('RIGHTPADDING', (0, 0), (-1, -1), 6)]
   176→   176→    for i in range(n):
   177→   177→        c = i * 2
   178→   178→        style.append(('BACKGROUND', (c, 0), (c, 1), CARD_BG))
   179→   179→        style.append(('BOX', (c, 0), (c, 1), 1, ACCENT))
   180→   180→    t.setStyle(TableStyle(style))
   181→   181→    return [Spacer(1, 8), t, Spacer(1, 14)]

   183→   183→def styled_table(data_rows, ratios, header_row=True, align_center_cols=None):
   184→   184→    """data_rows: primeira linha = header (texto simples). Celulas viram Paragraph."""
   185→   185→    align_center_cols = align_center_cols or []
   186→   186→    col_w = [r * AVAIL_W for r in ratios]
   187→   187→    assert abs(sum(ratios) - 1.0) < 0.01, 'ratios must sum to 1'
   188→   188→    wrapped = []
   189→   189→    for ri, row in enumerate(data_rows):
   190→   190→        out_row = []
   191→   191→        for ci, cell in enumerate(row):
   192→   192→            if ri == 0 and header_row:
   193→   193→                out_row.append(Paragraph('<b>%s</b>' % cell, th))
   194→   194→            else:
   195→   195→                st = tcc if ci in align_center_cols else tc
   196→   196→                out_row.append(Paragraph(str(cell), st))
   197→   197→        wrapped.append(out_row)
   198→   198→    t = Table(wrapped, colWidths=col_w, hAlign='CENTER', repeatRows=1)
   199→   199→    style = [('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   200→   200→             ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   201→   201→             ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   202→   202→             ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   203→   203→             ('LEFTPADDING', (0, 0), (-1, -1), 7),
   204→   204→             ('RIGHTPADDING', (0, 0), (-1, -1), 7),
   205→   205→             ('TOPPADDING', (0, 0), (-1, -1), 5.5),
   206→   206→             ('BOTTOMPADDING', (0, 0), (-1, -1), 5.5)]
   207→   207→    for ri in range(1, len(data_rows)):
   208→   208→        style.append(('BACKGROUND', (0, ri), (-1, ri),
   209→   209→                      TABLE_ROW_EVEN if ri % 2 == 1 else TABLE_ROW_ODD))
   210→   210→    t.setStyle(TableStyle(style))
   211→   211→    return t

   213→   213→def table_block(title_text, tbl, cap_text=None):
   214→   214→    out = [Spacer(1, 16)]
   215→   215→    title_p = Paragraph('<b>%s</b>' % title_text, h3)
   216→   216→    w, hh = tbl.wrap(AVAIL_W, PAGE_H)
   217→   217→    if hh <= MAX_KEEP:
   218→   218→        out += safe_keep([title_p, tbl])
   219→   219→    else:
   220→   220→        # tabela alta: nao agrupar - deixa fluir entre paginas (repeatRows repete cabecalho)
   221→   221→        out += [CondPageBreak(100), title_p, tbl]
   222→   222→    if cap_text:
   223→   223→        out += [Spacer(1, 4), Paragraph(cap_text, caption)]
   224→   224→    out += [Spacer(1, 14)]
   225→   225→    return out
